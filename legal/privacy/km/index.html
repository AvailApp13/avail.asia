គោលការណ៍ឯកជនភាពនៃកម្មវិធីទូរស័ព្ទ Avail

កំណែទី 4។ ចូលជាធរមាន: ថ្ងៃទី 24 ខែតុលា ឆ្នាំ 2026។

1. សេចក្តីផ្តើម

Avail (តទៅនេះ "វេទិកា", "យើង") គោរពឯកជនភាពរបស់អ្នក។ គោលការណ៍នេះពិពណ៌នាអំពីទិន្នន័យដែលយើងប្រមូល របៀបប្រើ អ្នកដែលយើងចែករំលែក និងសិទ្ធិរបស់អ្នក។ ការប្រើកម្មវិធីមានន័យថាអ្នកយល់ព្រមនឹងគោលការណ៍នេះ។

2. ទិន្នន័យដែលយើងប្រមូល

- ទិន្នន័យគណនី: ឈ្មោះ អាសយដ្ឋានអ៊ីមែល លេខទូរស័ព្ទ ប្រទេស និងទីក្រុង រូបថតប្រវត្តិរូប។ ពេលចូលតាម Apple ឬ Google យើងទទួលបានលេខសម្គាល់ និងទិន្នន័យដែលអ្នកអនុញ្ញាតឱ្យចែករំលែក។

- ខ្លឹមសារ: ការផ្សាយ និងរូបថត ការវាយតម្លៃ សារជជែក សំណើទៅផ្នែកគាំទ្រ។

- ទីតាំង: ដោយមានការអនុញ្ញាតរបស់អ្នក — សម្រាប់ស្វែងរកនៅជិត និងភ្ជាប់ការផ្សាយទៅផែនទី។ កូអរដោនេនៃការផ្សាយត្រូវបង្ហាញលើផែនទី; បើគ្មានអាសយដ្ឋានច្បាស់លាស់ ចំណុចនឹងបង្ហាញប្រហាក់ប្រហែល (ក្នុងសង្កាត់)។

- សំឡេង: ពេលប្រើការស្វែងរកដោយសំឡេង ឬដាក់ការផ្សាយដោយសំឡេង ការនិយាយរបស់អ្នកត្រូវបានសម្គាល់ដោយសេវាសម្គាល់ការនិយាយរបស់ឧបករណ៍ (Apple ឬ Google) ហើយការថតមិនត្រូវរក្សាទុកឡើយ។ អត្ថបទដែលសម្គាល់បានត្រូវដំណើរការដោយ AI ដើម្បីបំពេញវាលនៃការផ្សាយ ឬស្វែងរក។

- ការហៅក្នុងកម្មវិធី: ពេលហៅជាសំឡេងរវាងអ្នកប្រើ សំឡេងត្រូវបញ្ជូនតាមអ៊ីនធឺណិតដោយប្រើសេវាទំនាក់ទំនងពេលវេលាជាក់ស្តែងរបស់ភាគីទីបី។ ការសន្ទនាមិនត្រូវថត និងមិនត្រូវរក្សាទុកឡើយ។ យើងរក្សាកំណត់ហេតុការហៅ — អ្នកណាហៅអ្នកណា និងក្នុងការជជែកណា ម៉ោងចាប់ផ្តើម និងបញ្ចប់ និងស្ថានភាព (បានទទួល បានបដិសេធ ខកខាន); អ្នកចូលរួមការហៅអាចមើលវាបានក្នុងការជជែក។ ដើម្បីបញ្ជូនការហៅចូល យើងរក្សាទុក token បច្ចេកទេសរបស់ឧបករណ៍។ ការមើលតាមវីដេអូក៏ដូចគ្នា៖ តាមសំណើរបស់អ្នកទិញ អ្នកលក់បង្ហាញការផ្សាយតាមវីដេអូ វីដេអូមិនត្រូវថត ឬរក្សាទុកទេ ហើយរូបថតពីវីដេអូនៅតែលើទូរស័ព្ទរបស់អ្នកដែលថតប៉ុណ្ណោះ។

- ទិន្នន័យឧបករណ៍: ម៉ូដែល កំណែប្រព័ន្ធប្រតិបត្តិការ ភាសា លេខសម្គាល់ឧបករណ៍ token ការជូនដំណឹង push។

- ទិន្នន័យប្រើប្រាស់: ការមើល និងការស្វែងរក ការចុច "ហៅទូរស័ព្ទ" លើការផ្សាយ (ស្ថិតិសម្រាប់អ្នកលក់) កំណត់ហេតុកំហុសបច្ចេកទេស។

- ការទូទាត់: ព័ត៌មានការទិញតាម Apple App Store / Google Play (ការពិត និងខ្លឹមសារនៃការទិញ)។ យើងមិនទទួល និងមិនរក្សាទុកទិន្នន័យកាតរបស់អ្នកឡើយ។

3. របៀបយើងប្រើទិន្នន័យ

- ដំណើរការសេវាកម្ម: គណនី ការផ្សព្វផ្សាយ និងបង្ហាញការផ្សាយ ការជជែក និងការហៅ ការវាយតម្លៃ សេវាកម្មបង់ថ្លៃ និងសមតុល្យផ្ទៃក្នុង។

- សុវត្ថិភាព និងការត្រួតពិនិត្យ: ពិនិត្យការផ្សាយ (រួមទាំងការពិនិត្យស្វ័យប្រវត្តិដោយ AI) ដោះស្រាយបណ្តឹង ការពារការបោកប្រាស់។

- ការជូនដំណឹង: ការជូនដំណឹង push អំពីសារ ការត្រួតពិនិត្យ និងព្រឹត្តិការណ៍; អាចបិទក្នុងការកំណត់។

- ការកែលម្អសេវាកម្ម: វិភាគការប្រើប្រាស់ និងកំហុស។

4. អ្វីដែលអ្នកប្រើប្រាស់ផ្សេងឃើញ

ឈ្មោះ រូបថតប្រវត្តិរូប ទីក្រុង ចំណាត់ថ្នាក់ ការវាយតម្លៃ កាលបរិច្ឆេទចុះឈ្មោះ និងការផ្សាយសកម្មរបស់អ្នក អាចមើលឃើញដោយអ្នកប្រើប្រាស់ទាំងអស់។ ការដាក់ផ្សាយមានន័យថាអ្នកយល់ព្រមថាលេខទូរស័ព្ទរបស់អ្នកអាចប្រើដោយអ្នកប្រើប្រាស់ដើម្បីទាក់ទងតាមប៊ូតុង "ហៅទូរស័ព្ទ"។ សារជជែកអាចមើលឃើញដោយដៃគូសន្ទនារបស់អ្នក។

5. អ្នកដែលយើងចែករំលែកទិន្នន័យ

យើងចែករំលែកទិន្នន័យតែក្នុងកម្រិតចាំបាច់សម្រាប់ដំណើរការសេវាកម្មប៉ុណ្ណោះ ទៅកាន់ប្រភេទអ្នកទទួលដូចខាងក្រោម:

- អ្នកផ្តល់ហេដ្ឋារចនាសម្ព័ន្ធពពក — ម៉ាស៊ីនមេទិន្នន័យ និងឯកសារ;
- សេវាកម្មផ្ញើការជូនដំណឹង push;
- Apple / Google — ពេលដំណើរការការទិញក្នុងកម្មវិធី និងការសម្គាល់ការនិយាយ;
- សេវាទំនាក់ទំនងពេលវេលាជាក់ស្តែង — ការបញ្ជូនសំឡេងពេលហៅក្នុងកម្មវិធី;
- សេវាកម្មស្វ័យប្រវត្តិភាគីទីបី រួមទាំង AI — ការបកប្រែសារ ការសម្គាល់អត្ថបទការផ្សាយពេលផ្ទុកឡើងច្រើន ការស្វែងរកឆ្លាតវៃ ការត្រួតពិនិត្យ;
- សេវាកម្មកំណត់កូអរដោនេ និងផែនទី — ការកំណត់កូអរដោនេតាមអាសយដ្ឋានការផ្សាយ និងការបង្ហាញផែនទី។

អ្នកទទួលទាំងអស់ដំណើរការទិន្នន័យតាមការចាត់តាំងរបស់យើង និងសម្រាប់គោលបំណងដំណើរការសេវាកម្មប៉ុណ្ណោះ។ ទិន្នន័យអាចត្រូវផ្តល់ឱ្យអាជ្ញាធរអនុវត្តច្បាប់ និងស្ថាប័នមានសមត្ថកិច្ចផ្សេងទៀត ក្នុងករណីដែលច្បាប់កំណត់។ យើងមិនលក់ទិន្នន័យផ្ទាល់ខ្លួនរបស់អ្នកឡើយ។

6. រយៈពេលរក្សាទុក

ទិន្នន័យត្រូវរក្សាទុកខណៈគណនីរបស់អ្នកសកម្ម។ បន្ទាប់ពីលុបគណនី ទិន្នន័យត្រូវលុប ឬធ្វើអនាមិក; កំណត់ត្រាមួយចំនួន (ប្រវត្តិទូទាត់ កំណត់ត្រាការបំពាន) អាចរក្សាទុកយូរជាង បើច្បាប់តម្រូវ ឬដើម្បីការពារសិទ្ធិរបស់យើង។

7. សិទ្ធិរបស់អ្នក

- កែប្រែទិន្នន័យប្រវត្តិរូបក្នុងកម្មវិធី។
- លុបគណនីដោយខ្លួនឯង: ប្រវត្តិរូប → លុបគណនី។ សមតុល្យផ្ទៃក្នុងដែលមិនបានប្រើនឹងបាត់បង់។
- បិទការជូនដំណឹងក្នុងការកំណត់; ដកការអនុញ្ញាតទីតាំង និងមីក្រូហ្វូនក្នុងការកំណត់ឧបករណ៍។
- ស្នើសុំចូលមើល កែតម្រូវ ឬលុបទិន្នន័យរបស់អ្នកតាម info@avail.asia។

8. សុវត្ថិភាព

យើងអនុវត្តវិធានការការពារផ្នែករៀបចំ និងបច្ចេកទេសសមហេតុផល: ការអ៊ិនគ្រីបពេលបញ្ជូន ការគ្រប់គ្រងសិទ្ធិចូលប្រើកម្រិតមូលដ្ឋានទិន្នន័យ ការកំណត់សិទ្ធិបុគ្គលិក។

9. អនីតិជន

កម្មវិធីសម្រាប់បុគ្គលអាយុលើសពី 18 ឆ្នាំ។ យើងមិនប្រមូលទិន្នន័យអនីតិជនដោយចេតនាឡើយ; គណនីបែបនេះនឹងត្រូវលុប។

10. ការផ្លាស់ប្តូរគោលការណ៍

យើងអាចធ្វើបច្ចុប្បន្នភាពគោលការណ៍នេះ។ កំណែថ្មីចូលជាធរមានចាប់ពីពេលផ្សព្វផ្សាយក្នុងកម្មវិធី; ការផ្លាស់ប្តូរសំខាន់ៗនឹងជូនដំណឹងបន្ថែម។ ការបន្តប្រើកម្មវិធីមានន័យថាយល់ព្រមនឹងគោលការណ៍ដែលបានធ្វើបច្ចុប្បន្នភាព។
